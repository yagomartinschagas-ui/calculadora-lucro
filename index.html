function criarAbasEstruturadas() { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Aba de Precificação (Calculadora Geral) var abaPrecificacao = ss.getSheetByName("Precificação") || ss.insertSheet("Precificação"); abaPrecificacao.clear(); var dadosPrecificacao = [ ["CALCULADORA DE PRECIFICAÇÃO E LUCRO", "", "", ""], ["", "", "", ""], ["Produto / Descrição", "Custo de Produção / Aquisição (R$)", "Margem de Lucro Desejada (%)", "Preço de Venda Sugerido (R$)"], ["Exemplo: Embalagem P", 5.00, 100, "=B4*(1+(C4/100))"], ["", "", "", ""] ]; abaPrecificacao.getRange(1, 1, dadosPrecificacao.length, dadosPrecificacao[0].length).setValues(dadosPrecificacao); abaPrecificacao.getRange("A1:D1").merge().setBackground("#1155cc").setFontColor("#ffffff").setFontWeight("bold"); abaPrecificacao.getRange("A3:D3").setBackground("#d9d9d9").setFontWeight("bold"); // 2. Aba de Vendas e Custos (Fluxo Operacional) var abaVendas = ss.getSheetByName("Vendas e Custos") || ss.insertSheet("Vendas e Custos"); abaVendas.clear(); var dadosVendas = [ ["CONTROLE DE VENDAS E CUSTOS", "", "", "", "", "", ""], ["", "", "", "", "", "", ""], ["Data", "Produto", "Quantidade de Vendas", "Custo Unitário (R$)", "Preço de Venda Unitário (R$)", "Taxas Plataforma (R$)", "Lucro Total (R$)"], ["", "", "", "", "", "", "=IF(C4<>\"\"; (E4*C4) - (D4*C4) - F4; \"\")"], ["", "", "", "", "", "", "=IF(C5<>\"\"; (E5*C5) - (D5*C5) - F5; \"\")"] ]; abaVendas.getRange(1, 1, dadosVendas.length, dadosVendas[0].length).setValues(dadosVendas); abaVendas.getRange("A1:G1").merge().setBackground("#38761d").setFontColor("#ffffff").setFontWeight("bold"); abaVendas.getRange("A3:G3").setBackground("#d9d9d9").setFontWeight("bold"); // 3. Aba de Calculadora de ROAS (Ads) var abaRoas = ss.getSheetByName("Calculadora de ROAS") || ss.insertSheet("Calculadora de ROAS"); abaRoas.clear(); var dadosRoas = [ ["CALCULADORA DE ROAS (ANÚNCIOS - SHOPEE / MERCADO LIVRE)", "", "", ""], ["", "", "", ""], ["Métrica", "Valor", "", "Resultado / Análise"], ["Valor Investido em Ads (R$)", 100.00, "", "ROAS Obtido:"], ["Receita Gerada com Vendas (R$)", 500.00, "", "=IF(B4>0; B5/B4; 0)"], ["", "", "", ""], ["Custo por Aquisição - CPA (R$)", "=IF(B8>0; B4/B8; 0)", "", "ROAS de Equilíbrio (Mínimo):"], ["Quantidade de Vendas por Ads", 10, "", "Insira quanto precisa para não ter prejuízo"] ]; abaRoas.getRange(1, 1, dadosRoas.length, dadosRoas[0].length).setValues(dadosRoas); abaRoas.getRange("A1:D1").merge().setBackground("#b45f06").setFontColor("#ffffff").setFontWeight("bold"); abaRoas.getRange("A3:D3").setBackground("#d9d9d9").setFontWeight("bold"); SpreadsheetApp.getUi().alert("Abas criadas e organizadas com sucesso!"); }
