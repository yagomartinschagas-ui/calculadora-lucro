function organizarCalculadora() { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Aba de Resumo, Lucros, Despesas e Calculadora de Frete/Repasse var abaGestao = ss.getSheetByName("Gestão e Custos"); if (!abaGestao) { abaGestao = ss.insertSheet("Gestão e Custos"); } abaGestao.getRange("A1").setValue("=== CALCULADORA DE LUCRO E DESPESAS ==="); abaGestao.getRange("A3").setValue("Descrição"); abaGestao.getRange("B3").setValue("Valor / Detalhes"); abaGestao.getRange("D1").setValue("=== CALCULADORA DE ROUAS / FRETE (Shopee & Mercado Livre) ==="); abaGestao.getRange("D3").setValue("Plataforma"); abaGestao.getRange("E3").setValue("Taxas / Custos"); // 2. Aba de Vendas e Quantidades var abaVendas = ss.getSheetByName("Registro de Vendas"); if (!abaVendas) { abaVendas = ss.insertSheet("Registro de Vendas"); } abaVendas.getRange("A1").setValue("=== CONTROLE DE VENDAS E QUANTIDADES ==="); abaVendas.getRange("A3").setValue("Data"); abaVendas.getRange("B3").setValue("Produto"); abaVendas.getRange("C3").setValue("Quantidade"); abaVendas.getRange("D3").setValue("Custo Unitário"); abaVendas.getRange("E3").setValue("Preço de Venda"); SpreadsheetApp.getUi().alert("Calculadora organizada com sucesso em abas separadas!"); }
